# Объектив — план реализации

> **Для агентов:** ОБЯЗАТЕЛЬНЫЙ ПОД-СКИЛЛ: используйте superpowers:subagent-driven-development (рекомендуется) или superpowers:executing-plans, чтобы выполнять план задача за задачей. Шаги размечены чекбоксами (`- [ ]`).

**Цель:** система, которая подключает календарный график стройки по ссылке, сверяет его с детекциями техники и людей на снимках с камер и показывает работы, где отчётность подрядчика расходится с наблюдаемым фактом.

**Архитектура:** четыре независимых компонента поверх SQLite — импортёр плана (читает таблицу по ссылке, хранит версии), приёмник детекций (не знает про план), движок сверки (единственный, кто знает и про план, и про наблюдения) и HTTP API с одностраничным интерфейсом. Вывод идёт от плана к наблюдению: план говорит, что должно происходить, система проверяет, подтверждается ли это.

**Стек:** Python 3.11, FastAPI, sqlite3 из стандартной библиотеки, pytest, ultralytics YOLO для детекции, ванильный JavaScript для интерфейса.

## Глобальные ограничения

- Направление вывода: от плана к наблюдению. Классификация этапа по снимку не реализуется ни в каком виде.
- Уровни подтверждения в MVP: 1 (ресурсы на месте), 2 (работа идёт), 4 (работа принята по отчётности). Уровень 3 (рост объёма) не реализуется.
- Классы детекции: `excavator`, `dump_truck`, `crane`, `concrete_pump`, `mixer_truck`, `person`. Других классов в коде быть не должно.
- Порог уверенности детекции: `0.5`. Ниже — объект не засчитан.
- Порог смещения для уровня 2: `0.05` доли ширины кадра.
- Наблюдения сверяются с версией плана, действовавшей в день съёмки. Не с последней.
- Обязательные колонки плана: `work_id`, `name`, `zone`, `plan_start`, `plan_end`, `fact_percent`. Отсутствие любой — отказ импорта целиком, частичный импорт запрещён.
- Даты в БД хранятся строками ISO (`2026-09-04`), время — ISO с зоной.
- Статусы работы ровно семь: `reported_mismatch`, `not_confirmed`, `resources_only`, `confirmed`, `in_progress`, `not_started`, `not_checked`.
- Один объект. Мультиобъектности, ролей и авторизации нет.

```python
"""Подключение к SQLite и схема базы."""
from __future__ import annotations

import sqlite3
from pathlib import Path

SCHEMA = """
CREATE TABLE IF NOT EXISTS plan_versions (
    version_id INTEGER PRIMARY KEY AUTOINCREMENT,
    synced_at  TEXT NOT NULL,
    source_url TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS plan_works (
    version_id    INTEGER NOT NULL,
    work_id       TEXT NOT NULL,
    name          TEXT NOT NULL,
    zone          TEXT NOT NULL,
    plan_start    TEXT NOT NULL,
    plan_end      TEXT NOT NULL,
    fact_percent  REAL NOT NULL,
    contractor    TEXT,
    change_reason TEXT,
    PRIMARY KEY (version_id, work_id)
);

CREATE TABLE IF NOT EXISTS plan_changes (
    version_id INTEGER NOT NULL,
    work_id    TEXT NOT NULL,
    changed_at TEXT NOT NULL,
    change     TEXT NOT NULL,
    reason     TEXT
);

CREATE TABLE IF NOT EXISTS cameras (
    camera_id TEXT PRIMARY KEY,
    zone      TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS frames (
    frame_id    TEXT PRIMARY KEY,
    camera_id   TEXT NOT NULL,
    captured_at TEXT NOT NULL
);

CREATE TABLE IF NOT EXISTS detections (
    frame_id   TEXT NOT NULL,
    cls        TEXT NOT NULL,
    confidence REAL NOT NULL,
    x INTEGER NOT NULL,
    y INTEGER NOT NULL,
    w INTEGER NOT NULL,
    h INTEGER NOT NULL
);

CREATE TABLE IF NOT EXISTS holidays (
    day TEXT PRIMARY KEY
);

CREATE INDEX IF NOT EXISTS idx_frames_camera_time ON frames (camera_id, captured_at);
CREATE INDEX IF NOT EXISTS idx_detections_frame ON detections (frame_id);
"""


def connect(path: str | Path) -> sqlite3.Connection:
    """Открыть базу. Строки возвращаются как sqlite3.Row."""
    conn = sqlite3.connect(str(path))
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")
    return conn


def init_schema(conn: sqlite3.Connection) -> None:
    """Создать таблицы, если их ещё нет. Повторный вызов безопасен."""
    conn.executescript(SCHEMA)
    conn.commit()
```

```python
"""Разбор календарного плана из CSV."""
from __future__ import annotations

import csv
import io
from dataclasses import dataclass
from datetime import date

REQUIRED_COLUMNS = ("work_id", "name", "zone", "plan_start", "plan_end", "fact_percent")


class PlanImportError(Exception):
    """План не может быть импортирован. Частичный импорт запрещён."""


@dataclass(frozen=True)
class Work:
    work_id: str
    name: str
    zone: str
    plan_start: date
    plan_end: date
    fact_percent: float
    contractor: str | None = None
    change_reason: str | None = None


def _clean(value: str | None) -> str | None:
    if value is None:
        return None
    stripped = value.strip()
    return stripped or None


def _parse_date(value: str | None, work_id: str, column: str) -> date:
    text = _clean(value)
    if text is None:
        raise PlanImportError(f"Работа {work_id}: колонка {column} пуста")
    try:
        return date.fromisoformat(text)
    except ValueError as exc:
        raise PlanImportError(
            f"Работа {work_id}: колонка {column} — дата «{text}» "
            f"не в формате ГГГГ-ММ-ДД"
        ) from exc


def parse_plan_csv(text: str) -> list[Work]:
    """Разобрать CSV календарного плана. При любой ошибке отклонить целиком."""
    reader = csv.DictReader(io.StringIO(text))
    header = reader.fieldnames or []

    missing = [c for c in REQUIRED_COLUMNS if c not in header]
    if missing:
        raise PlanImportError(
            "В таблице нет обязательных колонок: " + ", ".join(missing)
        )

    works: list[Work] = []
    seen: set[str] = set()

    for row in reader:
        work_id = _clean(row.get("work_id"))
        if work_id is None:
            continue  # полностью пустая строка в конце таблицы

        if work_id in seen:
            raise PlanImportError(f"Работа {work_id} встречается в таблице дважды")
        seen.add(work_id)

        name = _clean(row.get("name"))
        zone = _clean(row.get("zone"))
        if name is None or zone is None:
            raise PlanImportError(f"Работа {work_id}: не заполнены name или zone")

        plan_start = _parse_date(row.get("plan_start"), work_id, "plan_start")
        plan_end = _parse_date(row.get("plan_end"), work_id, "plan_end")
        if plan_end < plan_start:
            raise PlanImportError(
                f"Работа {work_id}: окончание {plan_end} раньше начала {plan_start}"
            )

        raw_fact = _clean(row.get("fact_percent"))
        try:
            fact_percent = float(raw_fact.replace(",", ".")) if raw_fact else 0.0
        except ValueError as exc:
            raise PlanImportError(
                f"Работа {work_id}: fact_percent «{raw_fact}» не число"
            ) from exc

        works.append(
            Work(
                work_id=work_id,
                name=name,
                zone=zone,
                plan_start=plan_start,
                plan_end=plan_end,
                fact_percent=fact_percent,
                contractor=_clean(row.get("contractor")),
                change_reason=_clean(row.get("change_reason")),
            )
        )

    if not works:
        raise PlanImportError("В таблице нет ни одной работы")

    return works
```

```python
def test_save_version_returns_incrementing_ids(conn):
```

```python
"""Версии календарного плана и журнал изменений.

Версии существуют не ради истории как таковой: без них подрядчик,
получив претензию, сдвинет даты в таблице, и зафиксированное
расхождение исчезнет.
"""
from __future__ import annotations

import sqlite3
from dataclasses import dataclass
from datetime import date, datetime

from obyektiv.plan.parser import Work


@dataclass(frozen=True)
class PlanChange:
    work_id: str
    change: str
    reason: str | None


def _row_to_work(row: sqlite3.Row) -> Work:
    return Work(
        work_id=row["work_id"],
        name=row["name"],
        zone=row["zone"],
        plan_start=date.fromisoformat(row["plan_start"]),
        plan_end=date.fromisoformat(row["plan_end"]),
        fact_percent=row["fact_percent"],
        contractor=row["contractor"],
        change_reason=row["change_reason"],
    )


def latest_version_id(conn: sqlite3.Connection) -> int | None:
    row = conn.execute("SELECT MAX(version_id) AS v FROM plan_versions").fetchone()
    return row["v"]


def works_of_version(conn: sqlite3.Connection, version_id: int) -> list[Work]:
    rows = conn.execute(
        "SELECT * FROM plan_works WHERE version_id = ? ORDER BY work_id",
        (version_id,),
    ).fetchall()
    return [_row_to_work(r) for r in rows]


def works_at(conn: sqlite3.Connection, on_date: date) -> list[Work]:
    """Версия плана, действовавшая в указанный день.

    Наблюдения сверяются именно с ней, а не с последней версией, —
    поэтому правка графика задним числом не отменяет расхождения.
    """
    row = conn.execute(
        "SELECT MAX(version_id) AS v FROM plan_versions WHERE date(synced_at) <= ?",
        (on_date.isoformat(),),
    ).fetchone()
    if row["v"] is None:
        return []
    return works_of_version(conn, row["v"])


def _days(delta: int) -> str:
    sign = "+" if delta > 0 else "−"
    return f"{sign}{abs(delta)} дня" if abs(delta) in (1, 2, 3, 4) else f"{sign}{abs(delta)} дней"


def diff_works(old: list[Work], new: list[Work]) -> list[PlanChange]:
    """Что изменилось между двумя версиями плана."""
    old_by_id = {w.work_id: w for w in old}
    changes: list[PlanChange] = []

    for work in new:
        before = old_by_id.get(work.work_id)
        if before is None:
            changes.append(
                PlanChange(
                    work_id=work.work_id,
                    change=f"работа добавлена: {work.plan_start} — {work.plan_end}",
                    reason=work.change_reason,
                )
            )
            continue

        parts: list[str] = []
        if work.plan_start != before.plan_start:
            parts.append("начало " + _days((work.plan_start - before.plan_start).days))
        if work.plan_end != before.plan_end:
            parts.append("окончание " + _days((work.plan_end - before.plan_end).days))
        if work.fact_percent != before.fact_percent:
            parts.append(
                f"факт {before.fact_percent:g} % → {work.fact_percent:g} %"
            )

        if parts:
            changes.append(
                PlanChange(
                    work_id=work.work_id,
                    change=", ".join(parts),
                    reason=work.change_reason,
                )
            )

    for work_id in old_by_id.keys() - {w.work_id for w in new}:
        changes.append(
            PlanChange(work_id=work_id, change="работа убрана из графика", reason=None)
        )

    return changes


def save_version(
    conn: sqlite3.Connection,
    works: list[Work],
    source_url: str,
    synced_at: datetime,
) -> int:
    """Сохранить новую версию плана и записать изменения в журнал."""
    previous_id = latest_version_id(conn)
    previous = works_of_version(conn, previous_id) if previous_id else []

    cursor = conn.execute(
        "INSERT INTO plan_versions (synced_at, source_url) VALUES (?, ?)",
        (synced_at.isoformat(), source_url),
    )
    version_id = cursor.lastrowid

    conn.executemany(
        "INSERT INTO plan_works (version_id, work_id, name, zone, plan_start,"
        " plan_end, fact_percent, contractor, change_reason)"
        " VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
        [
            (
                version_id,
                w.work_id,
                w.name,
                w.zone,
                w.plan_start.isoformat(),
                w.plan_end.isoformat(),
                w.fact_percent,
                w.contractor,
                w.change_reason,
            )
            for w in works
        ],
    )

    conn.executemany(
        "INSERT INTO plan_changes (version_id, work_id, changed_at, change, reason)"
        " VALUES (?, ?, ?, ?, ?)",
        [
            (version_id, c.work_id, synced_at.isoformat(), c.change, c.reason)
            for c in diff_works(previous, works)
        ],
    )

    conn.commit()
    return version_id


def changes_for(conn: sqlite3.Connection, work_id: str) -> list[dict]:
    """Журнал изменений по одной работе, от старых к новым."""
    rows = conn.execute(
        "SELECT changed_at, change, reason FROM plan_changes"
        " WHERE work_id = ? ORDER BY changed_at",
        (work_id,),
    ).fetchall()
    return [dict(r) for r in rows]
```

```python
"""Получение календарного плана по ссылке.

План подключается ссылкой, а не разовой загрузкой файла: на стройке
график меняется практически ежедневно, и загруженный файл устаревает
к концу дня.
"""
from __future__ import annotations

import re
import sqlite3
from dataclasses import dataclass, field
from datetime import datetime

import httpx

from obyektiv.plan.parser import PlanImportError, parse_plan_csv
from obyektiv.plan.versions import (
    PlanChange,
    diff_works,
    latest_version_id,
    save_version,
    works_of_version,
)

_SHEETS = re.compile(r"^(https://docs\.google\.com/spreadsheets/d/[^/]+)/")


@dataclass(frozen=True)
class SyncResult:
    version_id: int | None
    created: bool
    changes: list[PlanChange] = field(default_factory=list)
    error: str | None = None


def to_csv_export_url(url: str) -> str:
    """Ссылку на Google Таблицу превратить в ссылку выгрузки CSV."""
    match = _SHEETS.match(url)
    if match:
        return f"{match.group(1)}/export?format=csv"
    return url


def fetch_plan_text(url: str, timeout: float = 10.0) -> str:
    response = httpx.get(to_csv_export_url(url), timeout=timeout, follow_redirects=True)
    response.raise_for_status()
    return response.text


def sync_plan(
    conn: sqlite3.Connection,
    url: str,
    now: datetime,
    fetch=fetch_plan_text,
) -> SyncResult:
    """Прочитать таблицу и сохранить новую версию, если план изменился.

    Недоступный источник или неверная таблица не разрушают состояние:
    система продолжает работать на последней успешной версии.
    """
    current_id = latest_version_id(conn)

    try:
        text = fetch(url, timeout=10.0)
        works = parse_plan_csv(text)
    except PlanImportError as exc:
        return SyncResult(version_id=current_id, created=False, error=str(exc))
    except Exception as exc:  # сеть, таймаут, HTTP-статус
        return SyncResult(
            version_id=current_id,
            created=False,
            error=f"Источник недоступен: {exc}",
        )

    previous = works_of_version(conn, current_id) if current_id else []
    changes = diff_works(previous, works)

    if current_id is not None and not changes:
        return SyncResult(version_id=current_id, created=False)

    version_id = save_version(conn, works, url, now)
    return SyncResult(version_id=version_id, created=True, changes=changes)
```

```python
"""Справочник признаков: название работы → что должна увидеть модель.

Второй источник данных, равный по важности плану. В реальных графиках
техника не указана — там написано «Бетонирование перекрытия, 3–5 сентября»
и всё.
"""
from __future__ import annotations

import csv
import io
from dataclasses import dataclass

KNOWN_CLASSES = frozenset(
    {"excavator", "dump_truck", "crane", "concrete_pump", "mixer_truck", "person"}
)


class CatalogError(Exception):
    """Справочник не может быть разобран."""


@dataclass(frozen=True)
class CatalogEntry:
    work_pattern: str
    required_classes: dict[str, int]
    min_people: int
    required_days: int


def _parse_classes(raw: str, pattern: str) -> dict[str, int]:
    result: dict[str, int] = {}
    for chunk in raw.split(";"):
        chunk = chunk.strip()
        if not chunk:
            continue
        if ":" not in chunk:
            raise CatalogError(
                f"«{pattern}»: «{chunk}» должно быть в виде класс:количество"
            )
        name, _, count = chunk.partition(":")
        name = name.strip()
        if name not in KNOWN_CLASSES:
            raise CatalogError(
                f"«{pattern}»: класс «{name}» модель не отдаёт. "
                f"Допустимы: {', '.join(sorted(KNOWN_CLASSES))}"
            )
        try:
            result[name] = int(count)
        except ValueError as exc:
            raise CatalogError(f"«{pattern}»: «{count}» не число") from exc
    if not result:
        raise CatalogError(f"«{pattern}»: не указан ни один класс")
    return result


def parse_catalog_csv(text: str) -> list[CatalogEntry]:
    reader = csv.DictReader(io.StringIO(text))
    entries: list[CatalogEntry] = []

    for row in reader:
        pattern = (row.get("work_pattern") or "").strip().lower()
        if not pattern:
            continue
        entries.append(
            CatalogEntry(
                work_pattern=pattern,
                required_classes=_parse_classes(row.get("required_classes", ""), pattern),
                min_people=int(row.get("min_people") or 0),
                required_days=int(row.get("required_days") or 1),
            )
        )

    if not entries:
        raise CatalogError("Справочник пуст")
    return entries


def entry_for(entries: list[CatalogEntry], work_name: str) -> CatalogEntry | None:
    """Найти запись справочника по названию работы.

    При нескольких совпадениях выигрывает самый длинный шаблон:
    «армирование фундамента» точнее, чем «фундамент».
    """
    name = work_name.lower()
    matched = [e for e in entries if e.work_pattern in name]
    if not matched:
        return None
    return max(matched, key=lambda e: len(e.work_pattern))
```

```python
"""Контракт между детектором и остальной системой."""
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime

from obyektiv.catalog import KNOWN_CLASSES

CONFIDENCE_THRESHOLD = 0.5


@dataclass(frozen=True)
class Detection:
    cls: str
    confidence: float
    bbox: tuple[int, int, int, int]  # x, y, ширина, высота в пикселях кадра


@dataclass(frozen=True)
class Frame:
    frame_id: str
    camera_id: str
    captured_at: datetime
    detections: list[Detection]


def frame_from_dict(data: dict) -> Frame:
    """Разобрать ответ детектора. Детекции ниже порога отбрасываются."""
    detections = [
        Detection(
            cls=item["class"],
            confidence=float(item["confidence"]),
            bbox=tuple(int(v) for v in item["bbox"]),
        )
        for item in data.get("detections", [])
        if float(item["confidence"]) >= CONFIDENCE_THRESHOLD
        and item["class"] in KNOWN_CLASSES
    ]
    return Frame(
        frame_id=data["frame_id"],
        camera_id=data["camera_id"],
        captured_at=datetime.fromisoformat(data["captured_at"]),
        detections=detections,
    )
```

```python
"""Хранение кадров и детекций, привязка камер к зонам."""
from __future__ import annotations

import sqlite3
from datetime import date, datetime

from obyektiv.detection.schema import Detection, Frame


def set_camera_zone(conn: sqlite3.Connection, camera_id: str, zone: str) -> None:
    conn.execute(
        "INSERT INTO cameras (camera_id, zone) VALUES (?, ?)"
        " ON CONFLICT(camera_id) DO UPDATE SET zone = excluded.zone",
        (camera_id, zone),
    )
    conn.commit()


def zone_of(conn: sqlite3.Connection, camera_id: str) -> str | None:
    row = conn.execute(
        "SELECT zone FROM cameras WHERE camera_id = ?", (camera_id,)
    ).fetchone()
    return row["zone"] if row else None


def save_frame(conn: sqlite3.Connection, frame: Frame) -> None:
    """Записать кадр и его детекции. Повторная запись перезаписывает."""
    conn.execute(
        "INSERT INTO frames (frame_id, camera_id, captured_at) VALUES (?, ?, ?)"
        " ON CONFLICT(frame_id) DO UPDATE SET"
        " camera_id = excluded.camera_id, captured_at = excluded.captured_at",
        (frame.frame_id, frame.camera_id, frame.captured_at.isoformat()),
    )
    conn.execute("DELETE FROM detections WHERE frame_id = ?", (frame.frame_id,))
    conn.executemany(
        "INSERT INTO detections (frame_id, cls, confidence, x, y, w, h)"
        " VALUES (?, ?, ?, ?, ?, ?, ?)",
        [
            (frame.frame_id, d.cls, d.confidence, *d.bbox)
            for d in frame.detections
        ],
    )
    conn.commit()


def _detections_of(conn: sqlite3.Connection, frame_id: str) -> list[Detection]:
    rows = conn.execute(
        "SELECT cls, confidence, x, y, w, h FROM detections WHERE frame_id = ?"
        " ORDER BY confidence DESC",
        (frame_id,),
    ).fetchall()
    return [
        Detection(
            cls=r["cls"],
            confidence=r["confidence"],
            bbox=(r["x"], r["y"], r["w"], r["h"]),
        )
        for r in rows
    ]


def _row_to_frame(conn: sqlite3.Connection, row: sqlite3.Row) -> Frame:
    return Frame(
        frame_id=row["frame_id"],
        camera_id=row["camera_id"],
        captured_at=datetime.fromisoformat(row["captured_at"]),
        detections=_detections_of(conn, row["frame_id"]),
    )


def load_frame(conn: sqlite3.Connection, frame_id: str) -> Frame | None:
    row = conn.execute(
        "SELECT * FROM frames WHERE frame_id = ?", (frame_id,)
    ).fetchone()
    return _row_to_frame(conn, row) if row else None


def frames_for_zone(conn: sqlite3.Connection, zone: str, day: date) -> list[Frame]:
    """Кадры всех камер зоны за один день, по возрастанию времени.

    Камеры без привязки к зоне в сверке не участвуют.
    """
    rows = conn.execute(
        "SELECT f.* FROM frames f"
        " JOIN cameras c ON c.camera_id = f.camera_id"
        " WHERE c.zone = ? AND date(f.captured_at) = ?"
        " ORDER BY f.captured_at",
        (zone, day.isoformat()),
    ).fetchall()
    return [_row_to_frame(conn, r) for r in rows]
```

```python
"""Детекция объектов на кадрах.

Модель ничего не знает про план, зоны и этапы работ: она отвечает
только на вопрос «что на кадре».
"""
from __future__ import annotations

import re
from datetime import datetime, timedelta, timezone
from pathlib import Path

MSK = timezone(timedelta(hours=3))

# Предобученная YOLO на COCO даёт готовыми только эти два класса.
# Остальные требуют дообучения; до тех пор они просто не находятся.
COCO_TO_OBYEKTIV = {
    "person": "person",
    "truck": "dump_truck",
}

_NAME = re.compile(r"^(?P<camera>.+)_(?P<day>\d{8})_(?P<time>\d{4})$")


def frame_id_from_path(path: Path) -> tuple[str, str, datetime]:
    """Разобрать имя файла вида КАМ-01_20260904_1420.jpg."""
    match = _NAME.match(path.stem)
    if not match:
        raise ValueError(
            f"Имя файла {path.name} не в формате <камера>_<ГГГГММДД>_<ЧЧММ>"
        )
    day = match.group("day")
    hhmm = match.group("time")
    captured_at = datetime(
        int(day[0:4]), int(day[4:6]), int(day[6:8]),
        int(hhmm[0:2]), int(hhmm[2:4]),
        tzinfo=MSK,
    )
    return path.stem, match.group("camera"), captured_at


def load_model(weights: str = "yolov8n.pt"):
    from ultralytics import YOLO

    return YOLO(weights)


def detect_image(model, path: Path) -> list[dict]:
    """Прогнать кадр через модель и вернуть детекции в контракте системы."""
    detections: list[dict] = []

    for result in model(str(path), verbose=False):
        for box in result.boxes:
            coco_name = result.names[int(box.cls[0])]
            mapped = COCO_TO_OBYEKTIV.get(coco_name)
            if mapped is None:
                continue
            x1, y1, x2, y2 = (int(v) for v in box.xyxy[0])
            detections.append(
                {
                    "class": mapped,
                    "confidence": float(box.conf[0]),
                    "bbox": [x1, y1, x2 - x1, y2 - y1],
                }
            )

    return detections
```

```python
        )
        processed += 1

    print(f"обработано кадров: {processed}")
    return 0
```

```python
"""Уровень подтверждения работ за один день.

Присутствие техники не равно выполнению работ. Отсюда разделение:
уровень 1 — ресурсы на месте, уровень 2 — техника меняет положение
или состав, то есть работа действительно идёт.
"""
from __future__ import annotations

from collections import Counter

from obyektiv.catalog import CatalogEntry
from obyektiv.detection.schema import Frame

MOVE_THRESHOLD = 0.05  # доля ширины кадра
FRAME_WIDTH = 1920


def counts_in_frame(frame: Frame) -> dict[str, int]:
    return dict(Counter(d.cls for d in frame.detections))


def observed_counts(frames: list[Frame]) -> dict[str, int]:
    """Максимум по каждому классу за все кадры дня."""
    result: dict[str, int] = {}
    for frame in frames:
        for cls, count in counts_in_frame(frame).items():
            result[cls] = max(result.get(cls, 0), count)
    return result


def _classes_found(observed: dict[str, int], entry: CatalogEntry) -> list[str]:
    """Требуемые классы, найденные в нужном количестве."""
    return [
        cls for cls, minimum in entry.required_classes.items()
        if observed.get(cls, 0) >= minimum
    ]


def _fully_resourced(observed: dict[str, int], entry: CatalogEntry) -> bool:
    return (
        len(_classes_found(observed, entry)) == len(entry.required_classes)
        and observed.get("person", 0) >= entry.min_people
    )


def _centers(frame: Frame, cls: str) -> list[float]:
    return sorted(
        d.bbox[0] + d.bbox[2] / 2 for d in frame.detections if d.cls == cls
    )


def _activity_detected(
    frames: list[Frame], entry: CatalogEntry, frame_width: int
) -> bool:
    """Техника сместилась или изменился её состав между соседними кадрами.

    Сравниваются только кадры одной камеры. В зоне может стоять несколько
    камер с разным полем зрения, и техника, видимая на одной и не видимая
    на другой, иначе читалась бы как движение.
    """
    threshold = frame_width * MOVE_THRESHOLD

    by_camera: dict[str, list[Frame]] = {}
    for frame in frames:
        by_camera.setdefault(frame.camera_id, []).append(frame)

    for camera_frames in by_camera.values():
        ordered = sorted(camera_frames, key=lambda f: f.captured_at)
        for cls in entry.required_classes:
            series = [_centers(f, cls) for f in ordered]
            for before, after in zip(series, series[1:]):
                if len(before) != len(after):
                    return True
                if any(abs(a - b) > threshold for a, b in zip(before, after)):
                    return True
    return False


def day_level(
    frames: list[Frame], entry: CatalogEntry, frame_width: int = FRAME_WIDTH
) -> int | None:
    """Уровень подтверждения за день.

    None — кадров нет; 0 — не найден ни один ожидаемый класс;
    1 — найдены не все классы либо людей меньше минимума, либо движения нет;
    2 — найдено всё ожидаемое и техника меняет положение.
    """
    if not frames:
        return None

    observed = observed_counts(frames)
    if not _classes_found(observed, entry):
        return 0
    if not _fully_resourced(observed, entry):
        return 1

    if len(frames) >= 2 and _activity_detected(frames, entry, frame_width):
        return 2

    return 1
```

```python
"""Статус работы: разрыв между обещанным и подтверждённым."""
from __future__ import annotations

import sqlite3
from dataclasses import dataclass
from datetime import date, timedelta

from obyektiv.catalog import CatalogEntry, entry_for
from obyektiv.detection.ingest import frames_for_zone
from obyektiv.matching.levels import day_level, observed_counts
from obyektiv.plan.parser import Work
from obyektiv.plan.versions import works_at

MAX_EVIDENCE_FRAMES = 3


@dataclass(frozen=True)
class DayResult:
    day: date
    level: int | None
    working: bool


@dataclass(frozen=True)
class WorkReport:
    work: Work
    entry: CatalogEntry | None
    days: list[DayResult]
    status: str
    observed: dict[str, int]
    evidence_frames: list[str]
    forecast_delay_days: int


def holidays_from_db(conn: sqlite3.Connection) -> set[date]:
    rows = conn.execute("SELECT day FROM holidays").fetchall()
    return {date.fromisoformat(r["day"]) for r in rows}


def working_days(start: date, end: date, holidays: set[date]) -> list[date]:
    """Рабочие дни отрезка: пятидневка минус перечисленные праздники.

    Без этого система ругалась бы на субботы и мгновенно потеряла доверие.
    """
    result: list[date] = []
    current = start
    while current <= end:
        if current.weekday() < 5 and current not in holidays:
            result.append(current)
        current += timedelta(days=1)
    return result


def _max_level(days: list[DayResult]) -> int:
    levels = [d.level for d in days if d.level is not None]
    return max(levels) if levels else 0


def _confirmed_days(days: list[DayResult]) -> int:
    return sum(1 for d in days if d.level is not None and d.level >= 2)


def forecast_delay(
    days: list[DayResult], entry: CatalogEntry | None, work: Work, today: date
) -> int:
    """Грубая экстраполяция: сколько подтверждённых дней недобрано.

    У ещё не начавшейся работы отставания нет. Без этой проверки каждая
    будущая работа показывала бы отставание, равное своему порогу.
    """
    if entry is None or today < work.plan_start:
        return 0
    return max(0, entry.required_days - _confirmed_days(days))


def decide_status(
    days: list[DayResult],
    entry: CatalogEntry | None,
    work: Work,
    today: date,
) -> str:
    if entry is None:
        return "not_checked"

    # Наблюдений нет вовсе — это отсутствие данных, а не обвинение подрядчика.
    if days and all(d.level is None for d in days):
        return "not_checked"

    max_level = _max_level(days)
    confirmed = _confirmed_days(days)

    if work.fact_percent >= 100 and max_level < 1:
        return "reported_mismatch"
    if today > work.plan_end and max_level < 2:
        return "not_confirmed"
    if max_level == 1:
        return "resources_only"
    if confirmed >= entry.required_days:
        return "confirmed"
    if today < work.plan_start:
        return "not_started"
    return "in_progress"


def _pick_evidence(
    conn: sqlite3.Connection, work: Work, days: list[DayResult]
) -> list[str]:
    """Кадры-основания: дни с наименьшим подтверждением идут первыми."""
    ranked = sorted(
        (d for d in days if d.level is not None),
        key=lambda d: (d.level, d.day),
    )
    frame_ids: list[str] = []
    for day_result in ranked:
        for frame in frames_for_zone(conn, work.zone, day_result.day):
            frame_ids.append(frame.frame_id)
            if len(frame_ids) >= MAX_EVIDENCE_FRAMES:
                return frame_ids
    return frame_ids


def build_report(
    conn: sqlite3.Connection,
    work: Work,
    entries: list[CatalogEntry],
    today: date,
    holidays: set[date],
) -> WorkReport:
    entry = entry_for(entries, work.name)

    days: list[DayResult] = []
    observed: dict[str, int] = {}

    if entry is not None:
        last_day = min(work.plan_end, today)
        for day in working_days(work.plan_start, last_day, holidays):
            frames = frames_for_zone(conn, work.zone, day)
            days.append(DayResult(day=day, level=day_level(frames, entry), working=True))
            for cls, count in observed_counts(frames).items():
                observed[cls] = max(observed.get(cls, 0), count)

    return WorkReport(
        work=work,
        entry=entry,
        days=days,
        status=decide_status(days, entry, work, today),
        observed=observed,
        evidence_frames=_pick_evidence(conn, work, days),
        forecast_delay_days=forecast_delay(days, entry, work, today),
    )


def build_all_reports(
    conn: sqlite3.Connection,
    entries: list[CatalogEntry],
    today: date,
    holidays: set[date],
) -> list[WorkReport]:
    """Отчёты по всем работам версии плана, действовавшей сегодня."""
    return [
        build_report(conn, work, entries, today, holidays)
        for work in works_at(conn, today)
    ]
```

```python
        "plan_history": changes_for(conn, work.work_id),
        "required_days": entry.required_days if entry else None,
    }


def create_app(
    db_path: str,
    catalog_path: str,
    plan_url: str,
    today: date | None = None,
) -> FastAPI:
    app = FastAPI(title="Объектив")
    entries = parse_catalog_csv(Path(catalog_path).read_text(encoding="utf-8"))

    def now() -> date:
        return today or date.today()

    def open_db():
        conn = connect(db_path)
        init_schema(conn)
        return conn

    @app.get("/api/object")
    def object_summary() -> dict:
        conn = open_db()
        try:
            reports = build_all_reports(conn, entries, now(), holidays_from_db(conn))
            version_id = latest_version_id(conn)
            version = (
                conn.execute(
                    "SELECT synced_at, source_url FROM plan_versions WHERE version_id = ?",
                    (version_id,),
                ).fetchone()
                if version_id
                else None
            )
            statuses = [r.status for r in reports]
            return {
                "today": now().isoformat(),
                "total_works": len(reports),
                "confirmed": sum(1 for s in statuses if s in ("confirmed", "in_progress")),
                "unconfirmed": sum(
                    1 for s in statuses if s in (NOT_CONFIRMED, RESOURCES_ONLY)
                ),
                "mismatch": statuses.count(MISMATCH),
                "forecast_delay_days": max(
                    (r.forecast_delay_days for r in reports), default=0
                ),
                "plan": {
                    "version_id": version_id,
                    "synced_at": version["synced_at"] if version else None,
                    "source_url": version["source_url"] if version else None,
                    "version_count": conn.execute(
                        "SELECT COUNT(*) AS n FROM plan_versions"
                    ).fetchone()["n"],
                },
                "works": [_brief(r) for r in reports],
            }
        finally:
            conn.close()

    @app.get("/api/works/{work_id}")
    def work_detail(work_id: str) -> dict:
        conn = open_db()
        try:
            match = [w for w in works_at(conn, now()) if w.work_id == work_id]
            if not match:
                raise HTTPException(status_code=404, detail=f"Работа {work_id} не найдена")
            report = build_report(conn, match[0], entries, now(), holidays_from_db(conn))
            return _detail(conn, report)
        finally:
            conn.close()

    @app.get("/api/frames/{frame_id}")
    def frame_detail(frame_id: str) -> dict:
        conn = open_db()
        try:
            frame = load_frame(conn, frame_id)
            if frame is None:
                raise HTTPException(status_code=404, detail=f"Кадр {frame_id} не найден")
            return {**_frame_to_dict(frame), "zone": zone_of(conn, frame.camera_id)}
        finally:
            conn.close()

    @app.get("/api/cameras/{camera_id}/{day}")
    def camera_day(camera_id: str, day: str) -> dict:
        conn = open_db()
        try:
            zone = zone_of(conn, camera_id)
            if zone is None:
                raise HTTPException(
                    status_code=404, detail=f"Камера {camera_id} не привязана к зоне"
                )
            frames = [
                f
                for f in frames_for_zone(conn, zone, date.fromisoformat(day))
                if f.camera_id == camera_id
            ]
            return {
                "camera_id": camera_id,
                "zone": zone,
                "day": day,
                "frames": [_frame_to_dict(f) for f in frames],
            }
        finally:
            conn.close()

    @app.post("/api/sync")
    def sync() -> dict:
        conn = open_db()
        try:
            from datetime import datetime

            result = sync_plan(conn, plan_url, datetime.now().astimezone())
            return {
                "version_id": result.version_id,
                "created": result.created,
                "error": result.error,
                "changes": [
                    {"work_id": c.work_id, "change": c.change, "reason": c.reason}
                    for c in result.changes
                ],
            }
        finally:
            conn.close()

    web_dir = Path(config.WEB_DIR)
    if web_dir.is_dir():
        app.mount("/", StaticFiles(directory=str(web_dir), html=True), name="web")

    return app


app = create_app(config.DB_PATH, config.CATALOG_PATH, config.PLAN_URL)
```

Задачи 1–4 (план), 5 (справочник) и 6–7 (детекция) не зависят друг от друга после задачи 1 и могут идти параллельно разными людьми. Задачи 8–9 требуют 5 и 6. Задача 10 требует всего предыдущего. Задачи 11–12 требуют 10.
